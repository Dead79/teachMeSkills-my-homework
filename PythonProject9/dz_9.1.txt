import os
import shutil

def main():
    # Информация об ОС и текущем пути
    print(f"Имя ОС: {os.name}")
    print(f"Текущий путь: {os.getcwd()}")
    
    # Получаем список файлов в текущей папке (исключая папки)
    files = [f for f in os.listdir() if os.path.isfile(f)]
    
    # Создаем словарь для группировки файлов по расширениям
    extensions = {}
    for file in files:
        ext = os.path.splitext(file)[1].lower()  # получаем расширение
        if ext not in extensions:
            extensions[ext] = []
        extensions[ext].append(file)
    
    # Создаем папки для каждого расширения и перемещаем файлы
    for ext, file_list in extensions.items():
        if ext:  # пропускаем файлы без расширения
            folder_name = ext[1:] + "_files"  # убираем точку из расширения
            os.makedirs(folder_name, exist_ok=True)
            
            total_size = 0
            moved_count = 0
            
            for file in file_list:
                file_size = os.path.getsize(file)
                total_size += file_size
                shutil.move(file, os.path.join(folder_name, file))
                moved_count += 1
            
            # Выводим информацию о перемещенных файлах
            size_gb = total_size / (1024 ** 3)
            print(f"В папке '{folder_name}' перемещено {moved_count} файлов, "
                  f"их суммарный размер – {size_gb:.2f} гигабайт")
    
    # Переименовываем один файл в одной из папок
    for ext in extensions:
        if ext:
            folder_name = ext[1:] + "_files"
            if os.path.exists(folder_name):
                files_in_folder = os.listdir(folder_name)
                if files_in_folder:
                    old_name = files_in_folder[0]
                    new_name = "renamed_" + old_name
                    old_path = os.path.join(folder_name, old_name)
                    new_path = os.path.join(folder_name, new_name)
                    os.rename(old_path, new_path)
                    print(f"Файл {old_name} был переименован в {new_name}")
                    break

if __name__ == "__main__":
    main()
